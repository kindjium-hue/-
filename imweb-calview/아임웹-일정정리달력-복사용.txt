<script>
{
/* 청명 일정 정리 — 위에 있는 아임웹 캘린더가 쓰는 일정 자료를 그대로 읽어,
   «고른 날짜의 일정»을 카드로 보여 준다. 날짜는 아임웹 캘린더에서 누른 날짜를 따라간다.
   아임웹 일정 상세 창이 열리면 그 안에 «청명 작업현황» 폼을 끼워 넣는다.

   자료 출처:  POST /ajax/calendar_data.cm   보냄 board_code, start, end
               받음 [{ id, title, start, end, className }]                       */

const DATA_URL = "/ajax/calendar_data.cm";
const DAY_NAMES = ["일", "월", "화", "수", "목", "금", "토"];
const CAL_TIME = "._calendar_time";
const PREV_BTN = ".fc-prev-button, [class*=prev-button]";
const NEXT_BTN = ".fc-next-button, [class*=next-button]";
const TITLE_KEYS = ["subject", "title", "name"];
const BODY_KEYS = ["content", "contents", "memo", "desc"];
const CELL_CSS = "padding:7px 10px;border:1px solid #d8dee6";
const HEAD_CSS = `width:92px;background:#f2f5f9;font-weight:700;${CELL_CSS}`;
const MOBILE_MAX = 640;
const NARROW_MAX = 400;

const root = document.querySelector("[data-calview-root]");
const pick = (name) => root.querySelector(`[data-${name}]`);
const setting = (name) => String(root.getAttribute(`data-${name}`) || "").trim();
const listOf = (name) => {
  const parts = setting(name).split(",");
  const out = [];
  for (let i = 0; i < parts.length; i += 1) {
    const one = parts[i].trim();
    if (one !== "") out.push(one);
  }
  return out;
};

const works = listOf("works");
const owners = listOf("owners");
const states = listOf("states");

const store = {};
let boardCode = setting("board-code");
let chosen = "";
let span = "month";
let loading = false;
let ready = false;
let selfCall = false;

/* ------------------------------------------------------------------ 날짜 */

const pad = (value) => `0${value}`.slice(-2);
const keyOf = (date) => `${date.getFullYear()}-${pad(date.getMonth() + 1)}-${pad(date.getDate())}`;
const todayKey = () => keyOf(new Date());
const fromKey = (key) => {
  const p = key.split("-");
  return new Date(Number(p[0]), Number(p[1]) - 1, Number(p[2]));
};
const addDays = (date, n) => new Date(date.getFullYear(), date.getMonth(), date.getDate() + n);
const longDate = (key) => {
  const d = fromKey(key);
  return `${d.getFullYear()}년 ${d.getMonth() + 1}월 ${d.getDate()}일 (${DAY_NAMES[d.getDay()]})`;
};
const shortDate = (key) => {
  const d = fromKey(key);
  return `${d.getMonth() + 1}/${d.getDate()} (${DAY_NAMES[d.getDay()]})`;
};
const datePart = (text) => {
  const found = String(text || "").match(/(\d{4})-(\d{1,2})-(\d{1,2})/);
  return found ? `${found[1]}-${pad(Number(found[2]))}-${pad(Number(found[3]))}` : "";
};
const timePart = (text) => {
  const found = String(text || "").match(/(\d{1,2}):(\d{2})/);
  return found ? `${pad(Number(found[1]))}:${found[2]}` : "";
};

/* -------------------------------------------------------------- 일정 자료 */

const colorClass = (name) => {
  const text = String(name || "");
  const found = text.match(/(\d+)/);
  if (found) return `c${Number(found[1]) % 6}`;
  if (text === "") return "c9";
  let sum = 0;
  for (let i = 0; i < text.length; i += 1) sum += text.charCodeAt(i);
  return `c${sum % 6}`;
};

const tagsIn = (title) => {
  const out = [];
  for (let i = 0; i < works.length; i += 1) {
    if (String(title || "").indexOf(works[i]) >= 0) out.push(works[i]);
  }
  return out;
};

const toItem = (row) => {
  const start = datePart(row.start);
  if (start === "") return null;
  let last = datePart(row.end) || start;
  const time = timePart(row.start);
  // 종일 일정의 끝날짜는 «다음 날»로 오는 관례가 있어 하루 당긴다
  if (last > start && timePart(row.end) === "" && time === "") last = keyOf(addDays(fromKey(last), -1));
  if (last < start) last = start;
  return {
    id: String(row.id || `${start}-${row.title}`),
    title: String(row.title || "").trim(),
    start: start,
    last: last,
    time: time,
    raw: `${row.start || ""} ~ ${row.end || ""}`,
    color: colorClass(row.className),
    tags: tagsIn(row.title),
  };
};

const keep = (rows) => {
  let added = 0;
  for (let i = 0; i < rows.length; i += 1) {
    const item = toItem(rows[i]);
    if (!item) continue;
    if (!store[item.id]) added += 1;
    store[item.id] = item;
  }
  return added;
};

const asRows = (data) => {
  if (Array.isArray(data)) return data;
  if (data && typeof data === "object") {
    const keys = ["data", "list", "value", "events", "result"];
    for (let i = 0; i < keys.length; i += 1) {
      if (Array.isArray(data[keys[i]])) return data[keys[i]];
    }
  }
  return [];
};

const onKey = (key) => {
  const out = [];
  const ids = Object.keys(store);
  for (let i = 0; i < ids.length; i += 1) {
    const item = store[ids[i]];
    if (item.start <= key && key <= item.last) out.push(item);
  }
  out.sort((a, b) => (a.time + a.title < b.time + b.title ? -1 : 1));
  return out;
};

const between = (from, to) => {
  const out = [];
  const ids = Object.keys(store);
  for (let i = 0; i < ids.length; i += 1) {
    const item = store[ids[i]];
    if (item.last >= from && item.start <= to) out.push(item);
  }
  out.sort((a, b) => (a.start + a.time < b.start + b.time ? -1 : 1));
  return out;
};

const say = (text, bad) => {
  const box = pick("status");
  box.textContent = text;
  box.hidden = text === "";
  if (bad) box.setAttribute("data-bad", "yes");
  else box.removeAttribute("data-bad");
};

const findBoardCode = () => {
  if (boardCode !== "") return boardCode;
  const found = document.documentElement.innerHTML
    .match(/board_code["'\s:=]{1,8}["']?([A-Za-z0-9_-]{8,40})/);
  if (found) boardCode = found[1];
  return boardCode;
};

const spanRange = () => {
  const at = chosen !== "" ? fromKey(chosen) : new Date();
  if (span === "year") {
    return [keyOf(new Date(at.getFullYear(), 0, 1)), keyOf(new Date(at.getFullYear(), 11, 31))];
  }
  if (span === "quarter") {
    return [keyOf(new Date(at.getFullYear(), at.getMonth() - 1, 1)),
      keyOf(new Date(at.getFullYear(), at.getMonth() + 2, 0))];
  }
  return [keyOf(new Date(at.getFullYear(), at.getMonth(), 1)),
    keyOf(new Date(at.getFullYear(), at.getMonth() + 1, 0))];
};

const ask = () => {
  const code = findBoardCode();
  if (code === "") {
    say("아임웹 캘린더를 찾지 못했습니다. 이 위젯을 캘린더와 같은 페이지에 두거나, "
      + "board_code 를 설정에 넣어 주세요.", true);
    return;
  }
  if (loading) return;
  loading = true;

  const bounds = spanRange();
  const from = keyOf(addDays(fromKey(bounds[0]), -7));
  const to = keyOf(addDays(fromKey(bounds[1]), 7));
  selfCall = true;
  const asking = fetch(DATA_URL, {
    method: "POST",
    credentials: "same-origin",
    headers: {
      "Content-Type": "application/x-www-form-urlencoded; charset=UTF-8",
      "X-Requested-With": "XMLHttpRequest",
    },
    body: `board_code=${encodeURIComponent(code)}&start=${from}&end=${to}`,
  });
  selfCall = false;
  asking
    .then((response) => {
      if (!response.ok) throw new Error(`${response.status}`);
      return response.json();
    })
    .then((data) => {
      loading = false;
      ready = true;
      keep(asRows(data));
      say("");
      draw();
    })
    .catch((error) => {
      loading = false;
      if (Object.keys(store).length === 0) {
        say(`일정을 받아오지 못했습니다 (${error.message || error}). `
          + "아임웹 캘린더와 같은 페이지인지 확인해 주세요.", true);
      }
      draw();
    });
};

/** 아임웹 캘린더가 스스로 부르는 요청을 엿봐 board_code 와 일정을 함께 얻는다 */
const listen = () => {
  const realOpen = XMLHttpRequest.prototype.open;
  const realSend = XMLHttpRequest.prototype.send;
  XMLHttpRequest.prototype.open = function (method, url) {
    this.calviewUrl = String(url || "");
    return realOpen.apply(this, arguments);
  };
  XMLHttpRequest.prototype.send = function (body) {
    if (String(this.calviewUrl || "").indexOf("calendar_data") >= 0) {
      const sent = typeof body === "string" ? body : "";
      const found = sent.match(/board_code=([^&]+)/);
      if (found && boardCode === "") boardCode = decodeURIComponent(found[1]);
      this.addEventListener("load", () => {
        let data = null;
        try { data = JSON.parse(this.responseText); } catch (error) { data = null; }
        if (!data) return;
        if (keep(asRows(data)) > 0 || !ready) {
          ready = true;
          say("");
          draw();
        }
      });
    }
    return realSend.apply(this, arguments);
  };

  const realFetch = window.fetch;
  if (typeof realFetch !== "function") return;
  window.fetch = function (input, init) {
    const url = typeof input === "string" ? input : (input && input.url) || "";
    const mine = String(url).indexOf("calendar_data") >= 0;
    const options = init || {};
    if (mine && typeof options.body === "string" && boardCode === "") {
      const found = options.body.match(/board_code=([^&]+)/);
      if (found) boardCode = decodeURIComponent(found[1]);
    }
    const skip = selfCall;
    return realFetch.apply(this, arguments).then((response) => {
      if (!mine || skip) return response;
      response.clone().json().then((data) => {
        if (keep(asRows(data)) > 0) {
          ready = true;
          say("");
          draw();
        }
      }, () => {});
      return response;
    });
  };
};

const shownMonth = () => {
  const el = document.querySelector(CAL_TIME);
  const found = el ? el.textContent.match(/(\d{4})\s*년\s*(\d{1,2})\s*월/) : null;
  return found ? `${found[1]}-${pad(Number(found[2]))}` : "";
};

const goMonth = (key, done) => {
  const want = key.slice(0, 7);
  let guard = 0;
  const step = () => {
    const now = shownMonth();
    if (now === "" || now === want || guard > 14) {
      done();
      return;
    }
    guard += 1;
    const button = document.querySelector(now > want ? PREV_BTN : NEXT_BTN);
    if (!button) {
      done();
      return;
    }
    button.click();
    setTimeout(step, 450);
  };
  step();
};

/* ---------------------------- 아임웹 상세 창에 «청명 작업현황» 폼 끼우기 */

const safe = (text) => String(text || "")
  .replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;");

const seen = (el) => !!(el && (el.offsetWidth > 0 || el.offsetHeight > 0));

/** 우리가 끼워 넣은 폼의 입력칸은 «채울 칸» 후보에서 빼야 한다 */
const ours = (el) => root.contains(el) || (el.closest && el.closest(".cvx-panel") !== null);

const matchKey = (el, keys) => {
  const name = `${el.getAttribute("name") || ""} ${el.getAttribute("id") || ""}`.toLowerCase();
  for (let i = 0; i < keys.length; i += 1) {
    if (name.indexOf(keys[i]) >= 0) return true;
  }
  return false;
};

const titleBox = (modal) => {
  const inputs = modal.querySelectorAll("input[type=text], input:not([type])");
  let first = null;
  for (let i = 0; i < inputs.length; i += 1) {
    if (!seen(inputs[i]) || ours(inputs[i])) continue;
    if (matchKey(inputs[i], TITLE_KEYS)) return inputs[i];
    if (!first) first = inputs[i];
  }
  return first;
};

const bodyBox = (modal) => {
  const areas = modal.querySelectorAll("textarea");
  for (let i = 0; i < areas.length; i += 1) {
    if (ours(areas[i])) continue;
    if (matchKey(areas[i], BODY_KEYS) || seen(areas[i])) return { kind: "area", el: areas[i] };
  }
  const rich = modal.querySelectorAll("[contenteditable=true], [contenteditable='']");
  for (let i = 0; i < rich.length; i += 1) {
    if (!ours(rich[i])) return { kind: "rich", el: rich[i] };
  }
  const frames = modal.querySelectorAll("iframe");
  for (let i = 0; i < frames.length; i += 1) {
    let doc = null;
    try { doc = frames[i].contentDocument; } catch (error) { doc = null; }
    if (doc && doc.body) return { kind: "frame", el: frames[i], doc: doc };
  }
  return null;
};

const panelData = (panel) => {
  const value = (name) => {
    const el = panel.querySelector(`[data-x-${name}]`);
    return el ? String(el.value || "").trim() : "";
  };
  const boxes = panel.querySelectorAll("[data-x-work]");
  const picked = [];
  for (let i = 0; i < boxes.length; i += 1) {
    if (boxes[i].checked) picked.push(boxes[i].getAttribute("data-x-work"));
  }
  const raw = value("note").split("\n");
  const note = [];
  for (let i = 0; i < raw.length; i += 1) {
    const line = raw[i].trim();
    if (line !== "") note.push(line);
  }
  return {
    works: picked,
    site: value("site"),
    place: value("place"),
    owner: value("owner"),
    state: value("state"),
    phone: value("phone"),
    note: note,
  };
};

const sheetHtml = (data) => {
  const rows = [];
  const row = (name, text) => {
    if (text === "") return;
    rows.push(`<tr><td style="${HEAD_CSS}">${name}</td><td style="${CELL_CSS}">${safe(text)}</td></tr>`);
  };
  row("작업항목", data.works.join(", "));
  row("현장", data.site);
  row("장소", data.place);
  row("담당자", data.owner);
  row("연락처", data.phone);
  row("진행상태", data.state);

  const out = [];
  if (rows.length > 0) {
    out.push('<table style="width:100%;border-collapse:collapse;font-size:14px;line-height:1.7">'
      + `<tbody>${rows.join("")}</tbody></table>`);
  }
  out.push('<p style="margin:16px 0 6px;font-weight:700">진행사항</p>');
  if (data.note.length > 0) {
    const items = [];
    for (let i = 0; i < data.note.length; i += 1) items.push(`<li>${safe(data.note[i])}</li>`);
    out.push(`<ul style="margin:0 0 14px;padding-left:20px;font-size:14px;line-height:1.8">${items.join("")}</ul>`);
  } else {
    out.push('<p style="margin:0 0 14px;color:#98a1ad;font-size:14px">작업이 진행될 때마다 적어 주세요.</p>');
  }
  out.push('<p style="margin:16px 0 6px;font-weight:700">BEFORE</p>');
  out.push('<p style="margin:0 0 14px;color:#98a1ad;font-size:14px">작업 전 사진을 이 자리에 넣어 주세요.</p>');
  out.push('<p style="margin:16px 0 6px;font-weight:700">AFTER</p>');
  out.push('<p style="margin:0 0 14px;color:#98a1ad;font-size:14px">작업 후 사진을 이 자리에 넣어 주세요.</p>');
  return out.join("");
};

const sheetText = (data) => {
  const out = [];
  const row = (name, text) => { if (text !== "") out.push(`${name}: ${text}`); };
  row("작업항목", data.works.join(", "));
  row("현장", data.site);
  row("장소", data.place);
  row("담당자", data.owner);
  row("연락처", data.phone);
  row("진행상태", data.state);
  out.push("");
  out.push("[진행사항]");
  if (data.note.length > 0) {
    for (let i = 0; i < data.note.length; i += 1) out.push(`- ${data.note[i]}`);
  } else {
    out.push("(작업이 진행될 때마다 적어 주세요)");
  }
  out.push("");
  out.push("[BEFORE]");
  out.push("(작업 전 사진)");
  out.push("");
  out.push("[AFTER]");
  out.push("(작업 후 사진)");
  return out.join("\n");
};

const titleLine = (data) => {
  const parts = [];
  if (data.site !== "") parts.push(data.site);
  if (data.works.length > 0) parts.push(data.works.join("·"));
  if (data.owner !== "") parts.push(data.owner);
  if (parts.length === 0) return "";
  return parts.join(" · ") + (data.state !== "" && data.state !== "예정" ? ` [${data.state}]` : "");
};

const putInto = (panel, modal, alsoTitle) => {
  const data = panelData(panel);
  const body = bodyBox(modal);
  const done = [];

  if (alsoTitle) {
    const box = titleBox(modal);
    const line = titleLine(data);
    if (box && line !== "") {
      box.value = line;
      box.dispatchEvent(new Event("input", { bubbles: true }));
      box.dispatchEvent(new Event("change", { bubbles: true }));
      done.push("제목");
    }
  }

  if (body) {
    if (body.kind === "area") {
      const now = String(body.el.value || "");
      body.el.value = now === "" ? sheetText(data) : `${now}\n\n${sheetText(data)}`;
      body.el.dispatchEvent(new Event("input", { bubbles: true }));
      body.el.dispatchEvent(new Event("change", { bubbles: true }));
    } else if (body.kind === "rich") {
      const wrap = document.createElement("div");
      wrap.innerHTML = sheetHtml(data);
      body.el.appendChild(wrap);
      body.el.dispatchEvent(new Event("input", { bubbles: true }));
    } else {
      const wrap = body.doc.createElement("div");
      wrap.innerHTML = sheetHtml(data);
      body.doc.body.appendChild(wrap);
    }
    done.push("본문");
  }

  if (done.length === 0) {
    tell(panel, "채울 칸을 찾지 못했습니다. «표 복사»를 눌러 본문에 직접 붙여넣어 주세요.");
    return;
  }
  tell(panel, `${done.join("·")}을 넣었습니다. 사진을 얹고 아임웹의 저장을 누르세요.`);
};

const filtered = () => {
  const bounds = spanRange();
  const query = String(pick("search").value || "").trim();
  const tag = pick("pick").value;
  const rows = between(bounds[0], bounds[1]);
  const out = [];

    const item = rows[i];
    if (query !== "" && item.title.indexOf(query) < 0) continue;

  }
  return out;
};

const drawSheet = () => {
  const rows = filtered();
  const body = pick("rows");
  body.textContent = "";

  const counts = {};
  for (let i = 0; i < rows.length; i += 1) {
    const tags = rows[i].tags;
    for (let t = 0; t < tags.length; t += 1) counts[tags[t]] = (counts[tags[t]] || 0) + 1;
  }
  const parts = [];
  const names = Object.keys(counts);
  for (let i = 0; i < names.length; i += 1) parts.push(`${names[i]} ${counts[names[i]]}`);
  const bounds = spanRange();
  pick("sum").textContent = `${bounds[0]} ~ ${bounds[1]} · 모두 ${rows.length}건`
    + (parts.length > 0 ? ` · ${parts.join(" · ")}` : "");

  if (rows.length === 0) {
    const tr = document.createElement("tr");
    const td = document.createElement("td");
    td.className = "cv-td";
    td.setAttribute("colspan", "4");
    td.textContent = "해당하는 일정이 없습니다.";
    tr.appendChild(td);
    body.appendChild(tr);
  }

  for (let i = 0; i < rows.length; i += 1) {
    const item = rows[i];
    const tr = document.createElement("tr");
    const date = document.createElement("td");
    date.className = "cv-td cv-td--date";
    date.textContent = item.start === item.last
      ? `${item.start}${item.time !== "" ? ` ${item.time}` : ""}`
      : `${item.start} ~ ${item.last}`;
    const name = document.createElement("td");
    name.className = "cv-td cv-td--name";
    name.textContent = item.title;
    const tags = document.createElement("td");
    tags.className = "cv-td";
    tags.textContent = item.tags.join(", ");
    const go = document.createElement("td");
    go.className = "cv-td";
    const open = document.createElement("button");
    open.className = "cv-open";
    open.setAttribute("type", "button");
    open.textContent = "열기";
    open.addEventListener("click", () => openItem(item));
    go.appendChild(open);
    tr.appendChild(date);
    tr.appendChild(name);
    tr.appendChild(tags);
    tr.appendChild(go);
    body.appendChild(tr);
  }

  const raw = pick("raw");
  raw.textContent = "";
  const all = between("0000-00-00", "9999-99-99");
  for (let i = 0; i < all.length && i < 40; i += 1) {
    const line = document.createElement("div");
    line.className = "cv-raw__line";
    line.textContent = `${all[i].title} — 받은 값: ${all[i].raw} → 읽은 날짜: ${all[i].start}`
      + (all[i].last !== all[i].start ? ` ~ ${all[i].last}` : "");
    raw.appendChild(line);
  }
};

const copySheet = () => {
  const rows = filtered();
  const lines = ["날짜\t일정\t작업항목"];
  for (let i = 0; i < rows.length; i += 1) {
    const item = rows[i];
    const when = item.start === item.last
      ? `${item.start}${item.time !== "" ? ` ${item.time}` : ""}`
      : `${item.start} ~ ${item.last}`;
    lines.push(`${when}\t${item.title}\t${item.tags.join(", ")}`);
  }
  const text = lines.join("\n");
  if (navigator.clipboard && navigator.clipboard.writeText) {
    navigator.clipboard.writeText(text).then(
      () => { pick("sum").textContent = `표 ${rows.length}줄을 복사했습니다. 엑셀에 붙여넣으세요.`; },
      () => { pick("sum").textContent = "복사가 막혔습니다. 표를 직접 선택해 주세요."; }
    );
    return;
  }
  pick("sum").textContent = "이 브라우저에서는 복사가 안 됩니다.";
};

const draw = () => {
  drawDay();
  drawSheet();
};

/* -------------------------------- 아임웹 달력에서 누른 날짜를 따라간다 */

const fromUrl = () => {
  const found = location.search.match(/[?&]d=(\d{4}-\d{1,2}-\d{1,2})/);
  return found ? datePart(found[1]) : "";
};

const followClicks = () => {
  document.addEventListener("click", (event) => {
    const target = event.target;
    if (!target || !target.closest || root.contains(target)) return;
    const cell = target.closest("[data-date]");
    if (!cell) return;
    const key = datePart(cell.getAttribute("data-date"));
    if (key === "" || key === chosen) return;
    chosen = key;
    draw();
    ask();
  }, true);
};

const setSpan = (mode) => {
  span = mode;
  pick("span-month").setAttribute("aria-pressed", mode === "month" ? "true" : "false");
  pick("span-quarter").setAttribute("aria-pressed", mode === "quarter" ? "true" : "false");
  pick("span-year").setAttribute("aria-pressed", mode === "year" ? "true" : "false");
  drawSheet();
  ask();
};

const applyWidth = () => {
  const width = root.offsetWidth || window.innerWidth || 0;
  if (width <= 0) return;
  if (width < MOBILE_MAX) root.classList.add("is-mobile");
  else root.classList.remove("is-mobile");
  if (width < NARROW_MAX) root.classList.add("is-narrow");
  else root.classList.remove("is-narrow");
};

const setup = () => {
  if (!root) return;
  listen();
  watchModal();
  followClicks();

  const box = pick("pick");
  const all = document.createElement("option");
  all.setAttribute("value", "");
  all.textContent = "작업항목 전체";
  box.appendChild(all);
  for (let i = 0; i < works.length; i += 1) {
    const option = document.createElement("option");
    option.setAttribute("value", works[i]);
    option.textContent = works[i];
    box.appendChild(option);
  }

  chosen = fromUrl() !== "" ? fromUrl() : todayKey();
  pick("reload").addEventListener("click", () => {
    ready = false;
    ask();
  });
  pick("span-month").addEventListener("click", () => setSpan("month"));
  pick("span-quarter").addEventListener("click", () => setSpan("quarter"));
  pick("span-year").addEventListener("click", () => setSpan("year"));
  pick("search").addEventListener("input", drawSheet);
  pick("pick").addEventListener("change", drawSheet);
  pick("copy").addEventListener("click", copySheet);

  draw();
  applyWidth();
  if (typeof ResizeObserver === "function") {
    const watcher = new ResizeObserver(() => applyWidth());
    watcher.observe(root);
  }
  window.addEventListener("resize", applyWidth);

  // 아임웹 캘린더가 먼저 자료를 받아오도록 조금 기다린 뒤 직접도 물어본다
  setTimeout(ask, 1200);
};

setup();
}
